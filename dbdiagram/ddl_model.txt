Table "inventory"."inventory_movements" {
  "inventory_movement_id" "INT IDENTITY(1,1)" [not null]
  "warehouse_id" INT [not null]
  "product_id" INT [not null]
  "batch_id" INT
  "movement_type" NVARCHAR(30) [not null]
  "movement_date" DATETIME2(3) [not null, default: `SYSUTCDATETIME()`]
  "quantity" DECIMAL(18,3) [not null]
  "unit_cost" DECIMAL(18,4) [not null]
  "total_cost" DECIMAL(19,4) [not null]
  "balance_quantity" DECIMAL(18,3) [not null]
  "balance_value" DECIMAL(19,4) [not null]
  "reference_type" NVARCHAR(50)
  "reference_id" INT
  "notes" NVARCHAR(500)
  "created_at" DATETIME2(3) [not null, default: `SYSUTCDATETIME()`]
  "created_by" INT [not null]

  Checks {
    `movement_type IN (
        'PURCHASE_RECEIPT',
        'SALE_DISPATCH',
        'TRANSFER_IN',
        'TRANSFER_OUT',
        'ADJUSTMENT_IN',
        'ADJUSTMENT_OUT',
        'RETURN_IN',
        'RETURN_OUT'
    )` [name: 'CK_inventory_movements_type']
    `(movement_type IN (
        'PURCHASE_RECEIPT',
        'TRANSFER_IN',
        'ADJUSTMENT_IN',
        'RETURN_IN'
    ) AND quantity > 0)
    OR
    (movement_type IN (
        'SALE_DISPATCH',
        'TRANSFER_OUT',
        'ADJUSTMENT_OUT',
        'RETURN_OUT'
    ) AND quantity < 0)` [name: 'CK_inventory_movements_quantity']
    `unit_cost >= 0
    AND total_cost = ROUND(quantity * unit_cost, 4)
    AND balance_quantity >= 0
    AND balance_value >= 0` [name: 'CK_inventory_movements_cost']
  }

  Indexes {
    inventory_movement_id [pk, name: "PK_inventory_movements"]
    (product_id, movement_date, inventory_movement_id) [name: "IX_inventory_movements_product_date"]
